create database app;
use app;

create table user (
    userId int auto_increment primary key,
    cpf char(11) unique not null,
    username varchar(50) not null,
    systemAdmin boolean default false
);



CREATE TABLE chat (
    chatId INT AUTO_INCREMENT PRIMARY KEY,
    chatType ENUM('DM', 'GROUP') NOT NULL,
    
    user1 INT,
    user2 INT,
    
    groupName VARCHAR(30),
    description VARCHAR(500),
    dtCreation DATE,
    creator int,

    FOREIGN KEY (creator) REFERENCES `user`(userId), 
    FOREIGN KEY (user1) REFERENCES `user`(userId),
    FOREIGN KEY (user2) REFERENCES `user`(userId),
    
    CONSTRAINT ck_chat_type CHECK (
        (chatType = 'DM' AND user1 IS NOT NULL AND user2 IS NOT NULL AND groupName IS NULL AND dtCreation IS NULL AND creator is NULL)
        OR
        (chatType = 'GROUP' AND user1 IS NULL AND user2 IS NULL AND groupName IS NOT NULL AND dtCreation IS NOT NULL AND creator is not null)
    )
);

create table userInChat (
    chatId int,
    userId int,
    entranceDay datetime default current_timestamp not null,
    foreign key (chatId) references chat (chatId),
    foreign key (userId) references user (userId),
    primary key (chatId, userId)
);

create table msg (
    msgId int auto_increment primary key,
    msgTxt varchar(500) not null,
    chatId int,
    sender int,
    sendDt datetime default current_timestamp,
    alterDt datetime default current_timestamp,
    foreign key (chatId) references chat (chatId),
    foreign key (sender) references user (userId)
);

create table hiddenMsg (
    msgId int ,
    userId int ,
    hideDt datetime default current_timestamp not null,
    foreign key (msgId) references msg (msgId),
    foreign key (userId) references user (userId),
    primary key (msgId, userId)
);

create table reports (
    userId int,
    chatId int,
    dtReport date not null,
    reason ENUM('SPAM', 'HATE', 'ILLEGAL'),
    primary key (userId, chatId),
    foreign key (userId) references user (userId),
    foreign key (chatId) references chat (chatId)
);

delimiter $$
create trigger personal_dm after insert on user 
for each row 
begin 
insert into chat (chatType, user1, user2) values ("DM", new.userId, new.userId); 
end$$
delimiter;
